เพิ่ม แก้ไขรูปภาพและข้อมูลไอเทม หรือตรวจสอบฐานข้อมูลสมาชิกจาก Google Sheet

เชื่อมต่อการบันทึกสมาชิกใหม่ลง Google Sheet อัตโนมัติ

โดยปกติหน้าเว็บจะอ่านข้อมูลจาก Google Sheet ได้ทันที แต่หากต้องการให้ระบบ สมัครสมาชิก เขียนแถวใหม่ลงใน Google Sheet (1-RTs9Vp7XVhRKVs4AL6YFMz3H_0oJKSCxyjev4vJGNc) โดยตรง ให้วางโค้ดด้านล่างลงใน Extensions > Apps Script ของชีตคุณ แล้วกด Deploy เป็น Web App (เลือก Who has access: Anyone) จากนั้นนำ URL มาวางที่นี่ค่ะ

โค้ด Google Apps Script (Code.gs) 
function doPost(e) {
  const sheet = SpreadsheetApp.openById('1-RTs9Vp7XVhRKVs4AL6YFMz3H_0oJKSCxyjev4vJGNc').getSheets()[0];
  const data = JSON.parse(e.postData.contents);
  if (data.action === 'register') {
    const rows = sheet.getDataRange().getValues();
    const norm = v => String(v || '').trim().replace(/\s+/g, ' ').toLowerCase();
    const exists = rows.some(r => norm(r[0]) === norm(data.username));
    if (exists) {
      return ContentService.createTextOutput(JSON.stringify({ status: 'duplicate' })).setMimeType(ContentService.MimeType.JSON);
    }
    // หากพบชื่อ-นามสกุลเดิมในชีต ให้อัปเดต Username/Password บนแถวนั้นพร้อมคงยอด GC เดิมไว้
    const matchIdx = rows.findIndex((r, idx) => idx > 0 && r.some(cell => norm(cell) === norm(data.fullName)));
    if (matchIdx > 0) {
      sheet.getRange(matchIdx + 1, 1).setValue(data.username);
      sheet.getRange(matchIdx + 1, 2).setValue(data.password);
      return ContentService.createTextOutput(JSON.stringify({ status: 'synced' })).setMimeType(ContentService.MimeType.JSON);
    }
    sheet.appendRow([data.username, data.password, data.gc || 0, data.role || 'member', data.fullName || '']);
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'invalid' })).setMimeType(ContentService.MimeType.JSON);
}
